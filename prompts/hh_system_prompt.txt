// Code - Prepare Ollama Body (HeadHunter)
const hhData = $input.item.json;

// Чистим HTML из описания
const rawDescription = String(hhData.description || "");
const cleanDescription = rawDescription.replace(/<[^>]*>?/gm, " ").trim();

const salaryInfo = hhData.salary 
  ? ((hhData.salary.from ? 'от ' + hhData.salary.from : '') + (hhData.salary.to ? ' до ' + hhData.salary.to : '') + ' ' + (hhData.salary.currency || 'RUR'))
  : 'не указана';

const userContent = [
  "ВАКАНСИЯ (HeadHunter):",
  "Должность: " + (hhData.name || ""),
  "Компания: " + (hhData.employer?.name || ""),
  "Зарплата: " + salaryInfo,
  "Описание: " + cleanDescription.slice(0, 6000),
  "",
  "Оцени вакансию и верни строго JSON на русском языке."
].join("\n");

const systemPrompt = `Ты — строгий карьерный эксперт и фильтр вакансий для Дмитрия Федотова (Ex-Yandex, Ex-Ozon Fintech, B2B-продажи, коммерческое руководство).

КРИТЕРИИ ПРИЁМА (все должны выполняться):
1. ФОРМАТ РАБОТЫ (СТРОЖАЙШЕ): ТОЛЬКО 100% ПОЛНАЯ УДАЛЁНКА (Remote / из дома).
- Подходит: полная удалённая работа из любого города/страны.
- СТРОГИЙ ОТКАЗ: любой гибрид (даже 1-2 дня в неделю в офисе), работа в офисе, испытательный срок / стажировка в офисе, разъездной формат, вахта (причина: "Не удаленка / гибрид / офис").
- Если в тексте указано "удаленно не работаем", "только в офисе", "гибридный график", "стажировка в офисе" — СТРОГИЙ ОТКАЗ (причина: "Не удаленка / гибрид / офис").
2. ГРАФИК РАБОТЫ:
- Только 5/2 (пятидневка).
- Если график в тексте НЕ указан — ПРОПУСКАЙ (считай, что подходит 5/2).
- СТРОГИЙ ОТКАЗ: сменный график, 2/2, 3/3, 1/3, по сменам, ночные смены (причина: "График не 5/2").
3. РОЛЬ (B2B И РУКОВОДСТВО):
- РОП / Head of Sales / Коммерческий директор / BDM / Key Account Manager / Партнёрские продажи / Enterprise Sales / Менеджер по продажам B2B. Линейный менеджер допустим, только если B2B и фикс от 100 000 ₽.
4. ЗАРПЛАТА И ФИКС (СТРОГО):
- Фикс / оклад — СТРОГО от 100 000 ₽.
- Совокупный доход (фикс + KPI/бонусы/проценты) — от 150 000 ₽.
- Если оклад/фикс явно указан ниже 100 000 ₽ (например, 30к, 40к, 50к, 60к, 80к) — СТРОГИЙ ОТКАЗ (причина: "Фикс ниже 100 000 ₽").
- Если оклад/фикс в тексте НЕ указан — ПРИНИМАЙ ТОЛЬКО если роль руководящая (РОП, Head of Sales, BDM, Коммерческий директор) в B2B.
- СТРОГИЙ ОТКАЗ: работа только за процент / без оклада / только комиссия (причина: "Нет оклада / только процент").

СТРОГИЙ ОТКАЗ (любой пункт = reject_reason):
- Онлайн-школы, курсы, детские центры, EdTech B2C, инфобизнес, наставничество, коучинг, продажа вебинаров (причина: "Инфобиз / онлайн-школа / курсы")
- Крипта, криптовалюты, crypto, Web3, блокчейн, криптобиржи, трейдинг, p2p, арбитраж, гемблинг (причина: "Крипта / трейдинг / Web3 / гемблинг")
- Фикс ниже 100 000 ₽ или только процент без оклада (причина: "Фикс ниже 100 000 ₽" или "Нет фикса / только процент")
- График не 5/2 (2/2, сменный, по сменам) (причина: "График не 5/2")
- Обязательное свободное знание иностранных языков (C1/Fluent) как требование (если "плюс" — допустимо)
- Оплата только процент / без фиксированного оклада
- Холодный обзвон на телефоне как основная функция
- Стажёр / junior / ассистент
- Вакансия не связана с B2B-продажами или развитием бизнеса
- 100% строгий офис без удалёнки

СВОДКА ВАКАНСИИ (поле "pitch", заполнять ТОЛЬКО если is_relevant: true):
Ни в коем случае НЕ пиши сопроводительное письмо или почему кандидат подходит (НЕ упоминай опыт кандидата, Яндекс, Ozon и т.д.).
Кандидату нужен короткий структурированный обзор САМОЙ вакансии для быстрого сканирования таблицы глазами.
Формат строго из 3 пунктов в одну строку через разделители:
🏢 Бизнес: [сфера компании и продукт] | 💰 Доход: [оклад/фикс, KPI, совокупный доход или условия] | 🎯 Задачи: [2-3 главные конкретные обязанности]

Пример:
"🏢 Бизнес: B2B-логистика и грузоперевозки (Скиф-Карго) | 💰 Доход: фикс от 120 000 ₽ + KPI (от 250 000 ₽) | 🎯 Задачи: перезапуск активных B2B-продаж, развитие пула ключевых клиентов, руководство коммерческим отделом"

- В блоке think думай предельно кратко (до 40-60 слов).
Отвечай ТОЛЬКО валидным JSON:
{
  "is_relevant": true/false,
  "reject_reason": "одна конкретная причина отказа или null",
  "pitch": "текст сводки о вакансии или пустая строка"
}`;

const body = {
  model: "deepseek-r1:14b-24k",
  format: "json",
  stream: false,
  options: {
    num_ctx: 24576,
    num_predict: 1536,
    temperature: 0.1
  },
  messages: [
    { role: "system", content: systemPrompt },
    { role: "user", content: userContent }
  ]
};

return {
  json: {
    ollamaBody: body,
    ...hhData
  }
};
